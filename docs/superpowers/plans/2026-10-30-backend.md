# CoworkingBooking v2 — Backend: plan de implementación

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Construir la API REST de reservas de coworking (espacios, usuarios con JWT y reservas sin solapamiento) en `backend/`, lista para empaquetarse en Docker.

**Architecture:** Spring Boot organizado por funcionalidad (`espacio`, `usuario`, `auth`, `reserva`, `common`), cada paquete con controller → service → repository. El esquema lo gestiona Flyway; Hibernate solo valida. Las reglas de horario viven en una clase pura (`ReglasHorario`) y el solapamiento se impide en el servicio y con una exclusion constraint de PostgreSQL.

**Tech Stack:** Java 21, Spring Boot 3.5.16, Spring Data JPA, Spring Security + oauth2-resource-server (JWT HS256), Flyway, PostgreSQL 16, springdoc-openapi 2.8.17, Lombok (solo entidades), JUnit 5, AssertJ, Mockito, Testcontainers, Docker Compose.

**Spec:** `docs/superpowers/specs/2026-10-30-backend-design.md`

## Global Constraints

- Raíz del repo: `/Users/david/Documents/proyectos_backend/coworking-booking`. Todo el código va en `backend/`. Los comandos se ejecutan desde la raíz.
- Maven se ejecuta siempre con JDK 21: `(cd backend && JAVA_HOME=$(/usr/libexec/java_home -v 21) ./mvnw <args>)`. El JDK por defecto de la máquina es 25: no usarlo.
- Docker debe estar corriendo (Testcontainers y Docker Compose).
- Paquete base: `co.devdavco.coworking`. groupId `co.devdavco`, artifactId `coworking-booking`.
- Versiones exactas: `spring-boot-starter-parent` 3.5.16, `springdoc-openapi-starter-webmvc-ui` 2.8.17, imagen `postgres:16-alpine`.
- Lombok solo en entidades JPA. Servicios y controllers usan constructores explícitos. DTOs son `record`.
- Entidades usan `Instant` para fechas. La API usa `OffsetDateTime` en la zona `app.reservas.zona-horaria` (`America/Bogota`).
- `spring.jpa.hibernate.ddl-auto=validate` y `spring.jpa.open-in-view=false`. El esquema solo cambia con migraciones Flyway nuevas; nunca editar una migración ya commiteada.
- Toda respuesta de error es `ProblemDetail` con estos `title` exactos: `Datos inválidos` (400), `Credenciales inválidas` (401 de login), `No autenticado` (401), `Acceso denegado` (403), `Recurso no encontrado` (404), `Conflicto` (409), `Regla de negocio incumplida` (422).
- Ninguna respuesta incluye la contraseña ni su hash.
- Textos para el usuario en español. Mensajes de commit en español, con el trailer `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.

## Review Focus

1. Contraseña con caracteres multibyte (`"ñ".repeat(40)`: 40 caracteres, 80 bytes) → 400, no 500 de BCrypt. Test en Tarea 4.
2. `horaInicio` sin zona horaria (`"2026-11-03T09:00:00"`) → 400, no interpretarla en la zona del servidor. Test en Tarea 7.
3. Reserva que empieza exactamente cuando termina la limpieza de otra → se acepta (rango `[inicio, finTotal)`). Test en Tarea 7.
4. El cliente envía `usuarioId` de otra persona en el cuerpo de la reserva → se ignora; la reserva queda a nombre del dueño del token. Test en Tarea 7.
5. Email con mayúsculas en registro y login (`Ana@Mail.com` / `ANA@mail.com`) → misma cuenta; duplicado detectado sin importar mayúsculas. Test en Tarea 4.

```sql
CREATE EXTENSION IF NOT EXISTS btree_gist;

CREATE TABLE usuarios (
    id            BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    nombre        VARCHAR(100) NOT NULL,
    email         VARCHAR(150) NOT NULL UNIQUE,
    password_hash VARCHAR(100) NOT NULL,
    rol           VARCHAR(20)  NOT NULL CHECK (rol IN ('ADMIN', 'USUARIO')),
    creado_en     TIMESTAMPTZ  NOT NULL DEFAULT now()
);

CREATE TABLE espacios (
    id               BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    nombre           VARCHAR(50) NOT NULL UNIQUE,
    tipo             VARCHAR(30) NOT NULL
                     CHECK (tipo IN ('ESCRITORIO', 'SALA_REUNIONES', 'SALA_CONFERENCIAS', 'CABINA')),
    capacidad        INT         NOT NULL CHECK (capacidad > 0),
    minutos_limpieza INT         NOT NULL CHECK (minutos_limpieza >= 0),
    activo           BOOLEAN     NOT NULL DEFAULT TRUE
);

CREATE TABLE reservas (
    id             BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    usuario_id     BIGINT      NOT NULL REFERENCES usuarios (id),
    espacio_id     BIGINT      NOT NULL REFERENCES espacios (id),
    hora_inicio    TIMESTAMPTZ NOT NULL,
    hora_fin       TIMESTAMPTZ NOT NULL,
    hora_fin_total TIMESTAMPTZ NOT NULL,
    estado         VARCHAR(20) NOT NULL CHECK (estado IN ('CONFIRMADA', 'CANCELADA')),
    version        INT         NOT NULL DEFAULT 0,
    creado_en      TIMESTAMPTZ NOT NULL DEFAULT now(),
    CONSTRAINT reservas_fin_despues_de_inicio CHECK (hora_fin > hora_inicio),
    CONSTRAINT reservas_sin_solapamiento
        EXCLUDE USING gist (espacio_id WITH =, tstzrange(hora_inicio, hora_fin_total, '[)') WITH &&)
        WHERE (estado = 'CONFIRMADA')
);

CREATE INDEX idx_reservas_usuario ON reservas (usuario_id);
CREATE INDEX idx_reservas_espacio_inicio ON reservas (espacio_id, hora_inicio);
```

```java
package co.devdavco.coworking.espacio;

import jakarta.persistence.Column;
import jakarta.persistence.Entity;
import jakarta.persistence.EnumType;
import jakarta.persistence.Enumerated;
import jakarta.persistence.GeneratedValue;
import jakarta.persistence.GenerationType;
import jakarta.persistence.Id;
import jakarta.persistence.Table;
import lombok.Getter;
import lombok.NoArgsConstructor;
import lombok.Setter;

@Entity
@Table(name = "espacios")
@Getter
@Setter
@NoArgsConstructor
public class Espacio {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    @Column(nullable = false, length = 50, unique = true)
    private String nombre;

    @Enumerated(EnumType.STRING)
    @Column(nullable = false, length = 30)
    private TipoEspacio tipo;

    @Column(nullable = false)
    private Integer capacidad;

    @Column(name = "minutos_limpieza", nullable = false)
    private Integer minutosLimpieza;

    @Column(nullable = false)
    private boolean activo = true;

    public Espacio(String nombre, TipoEspacio tipo, int capacidad, int minutosLimpieza) {
        this.nombre = nombre;
        this.tipo = tipo;
        this.capacidad = capacidad;
        this.minutosLimpieza = minutosLimpieza;
    }
}
```

```java
package co.devdavco.coworking.common;

import org.springframework.dao.DataIntegrityViolationException;
import org.springframework.http.HttpHeaders;
import org.springframework.http.HttpStatus;
import org.springframework.http.HttpStatusCode;
import org.springframework.http.ProblemDetail;
import org.springframework.http.ResponseEntity;
import org.springframework.orm.ObjectOptimisticLockingFailureException;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;
import org.springframework.web.context.request.WebRequest;
import org.springframework.web.servlet.mvc.method.annotation.ResponseEntityExceptionHandler;

import java.util.LinkedHashMap;
import java.util.Map;

/**
 * Traduce las excepciones a respuestas ProblemDetail (RFC 7807).
 * Extiende ResponseEntityExceptionHandler para que los errores de Spring MVC
 * (JSON mal formado, parámetros faltantes, tipos inválidos) también salgan como ProblemDetail.
 */
@RestControllerAdvice
public class GlobalExceptionHandler extends ResponseEntityExceptionHandler {

    @ExceptionHandler(RecursoNoEncontradoException.class)
    ProblemDetail noEncontrado(RecursoNoEncontradoException ex) {
        return problema(HttpStatus.NOT_FOUND, "Recurso no encontrado", ex.getMessage());
    }

    @ExceptionHandler(ConflictoException.class)
    ProblemDetail conflicto(ConflictoException ex) {
        return problema(HttpStatus.CONFLICT, "Conflicto", ex.getMessage());
    }

    @ExceptionHandler(ReglaNegocioException.class)
    ProblemDetail reglaNegocio(ReglaNegocioException ex) {
        return problema(HttpStatus.UNPROCESSABLE_ENTITY, "Regla de negocio incumplida", ex.getMessage());
    }

    @ExceptionHandler(ObjectOptimisticLockingFailureException.class)
    ProblemDetail versionDesactualizada(ObjectOptimisticLockingFailureException ex) {
        return problema(HttpStatus.CONFLICT, "Conflicto",
                "El recurso fue modificado por otra petición; vuelve a intentarlo");
    }

    @ExceptionHandler(DataIntegrityViolationException.class)
    ProblemDetail integridad(DataIntegrityViolationException ex) {
        return problema(HttpStatus.CONFLICT, "Conflicto", "La operación entra en conflicto con datos existentes");
    }

    @Override
    protected ResponseEntity<Object> handleMethodArgumentNotValid(
            MethodArgumentNotValidException ex, HttpHeaders headers, HttpStatusCode status, WebRequest request) {
        Map<String, String> errores = new LinkedHashMap<>();
        ex.getBindingResult().getFieldErrors()
                .forEach(error -> errores.putIfAbsent(error.getField(), error.getDefaultMessage()));

        ProblemDetail detalle = problema(HttpStatus.BAD_REQUEST, "Datos inválidos", "La petición tiene campos inválidos");
        detalle.setProperty("errores", errores);
        return ResponseEntity.badRequest().body(detalle);
    }

    private static ProblemDetail problema(HttpStatus status, String titulo, String detalle) {
        ProblemDetail problema = ProblemDetail.forStatusAndDetail(status, detalle);
        problema.setTitle(titulo);
        return problema;
    }
}
```

```java
package co.devdavco.coworking.espacio;

import co.devdavco.coworking.common.ConflictoException;
import co.devdavco.coworking.common.RecursoNoEncontradoException;
import co.devdavco.coworking.espacio.dto.EspacioRequest;
import co.devdavco.coworking.espacio.dto.EspacioResponse;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.List;

@Service
@Transactional(readOnly = true)
public class EspacioService {

    private final EspacioRepository espacios;

    public EspacioService(EspacioRepository espacios) {
        this.espacios = espacios;
    }

    public List<EspacioResponse> listarActivos() {
        return espacios.findAllByActivoTrueOrderByNombreAsc().stream()
                .map(EspacioResponse::desde)
                .toList();
    }

    public EspacioResponse obtener(Long id) {
        return EspacioResponse.desde(buscarActivo(id));
    }

    @Transactional
    public EspacioResponse crear(EspacioRequest request) {
        String nombre = request.nombre().trim();
        if (espacios.existsByNombreIgnoreCase(nombre)) {
            throw nombreDuplicado(nombre);
        }
        Espacio espacio = new Espacio(nombre, request.tipo(), request.capacidad(), request.minutosLimpieza());
        return EspacioResponse.desde(espacios.save(espacio));
    }

    @Transactional
    public EspacioResponse actualizar(Long id, EspacioRequest request) {
        Espacio espacio = buscarActivo(id);
        String nombre = request.nombre().trim();
        if (espacios.existsByNombreIgnoreCaseAndIdNot(nombre, id)) {
            throw nombreDuplicado(nombre);
        }
        espacio.setNombre(nombre);
        espacio.setTipo(request.tipo());
        espacio.setCapacidad(request.capacidad());
        espacio.setMinutosLimpieza(request.minutosLimpieza());
        return EspacioResponse.desde(espacio);
    }

    @Transactional
    public void desactivar(Long id) {
        buscarActivo(id).setActivo(false);
    }

    /** Devuelve la entidad; lo usan otros servicios (por ejemplo, reservas). */
    public Espacio buscarActivo(Long id) {
        return espacios.findByIdAndActivoTrue(id)
                .orElseThrow(() -> new RecursoNoEncontradoException("No existe el espacio " + id));
    }

    private static ConflictoException nombreDuplicado(String nombre) {
        return new ConflictoException("Ya existe un espacio llamado '" + nombre + "'");
    }
}
```

```java
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.DeleteMapping;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.PutMapping;

import org.springframework.web.bind.annotation.ResponseStatus;
import org.springframework.web.bind.annotation.RestController;

import java.util.List;

        this.espacioService = espacioService;
    }

    @GetMapping

        EspacioResponse creado = espacioService.crear(request);
        return ResponseEntity.created(URI.create("/api/espacios/" + creado.id())).body(creado);
    }

    @PutMapping("/{id}")
    public EspacioResponse actualizar(@PathVariable Long id, @Valid @RequestBody EspacioRequest request) {
        return espacioService.actualizar(id, request);
    }

    @DeleteMapping("/{id}")
    @ResponseStatus(HttpStatus.NO_CONTENT)
    public void desactivar(@PathVariable Long id) {

    }
}
```

```java
package co.devdavco.coworking.usuario;

import jakarta.persistence.Column;
import jakarta.persistence.Entity;
import jakarta.persistence.EnumType;
import jakarta.persistence.Enumerated;
import jakarta.persistence.GeneratedValue;
import jakarta.persistence.GenerationType;
import jakarta.persistence.Id;
import jakarta.persistence.Table;
import lombok.Getter;
import lombok.NoArgsConstructor;
import lombok.Setter;

import java.time.Instant;

@Entity
@Table(name = "usuarios")
@Getter
@Setter
@NoArgsConstructor
public class Usuario {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    @Column(nullable = false, length = 100)
    private String nombre;

    @Column(nullable = false, length = 150, unique = true)
    private String email;

    @Column(name = "password_hash", nullable = false, length = 100)
    private String passwordHash;

    @Enumerated(EnumType.STRING)
```

```java
package co.devdavco.coworking.common;

import com.fasterxml.jackson.databind.ObjectMapper;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import org.springframework.http.HttpStatus;
import org.springframework.http.MediaType;
import org.springframework.http.ProblemDetail;
import org.springframework.security.access.AccessDeniedException;
import org.springframework.security.core.AuthenticationException;
import org.springframework.security.web.AuthenticationEntryPoint;
import org.springframework.security.web.access.AccessDeniedHandler;
import org.springframework.stereotype.Component;

import java.io.IOException;
import java.net.URI;

/** Responde 401 y 403 con ProblemDetail, igual que el resto de errores de la API. */
@Component
public class ProblemaSeguridadHandler implements AuthenticationEntryPoint, AccessDeniedHandler {

    private final ObjectMapper objectMapper;

    public ProblemaSeguridadHandler(ObjectMapper objectMapper) {
        this.objectMapper = objectMapper;
    }

    @Override
    public void commence(HttpServletRequest request, HttpServletResponse response,
                         AuthenticationException ex) throws IOException {
        escribir(request, response, HttpStatus.UNAUTHORIZED, "No autenticado",
                "Necesitas un token válido para acceder a este recurso");
    }

    @Override
    public void handle(HttpServletRequest request, HttpServletResponse response,
                       AccessDeniedException ex) throws IOException {
        escribir(request, response, HttpStatus.FORBIDDEN, "Acceso denegado",
                "No tienes permiso para realizar esta operación");
    }

    private void escribir(HttpServletRequest request, HttpServletResponse response,
                          HttpStatus status, String titulo, String detalle) throws IOException {
        ProblemDetail problema = ProblemDetail.forStatusAndDetail(status, detalle);
        problema.setTitle(titulo);
        problema.setInstance(URI.create(request.getRequestURI()));
        response.setStatus(status.value());
        response.setContentType(MediaType.APPLICATION_PROBLEM_JSON_VALUE);
        response.setCharacterEncoding("UTF-8");
        objectMapper.writeValue(response.getOutputStream(), problema);
    }
}
```

```java
                        .requestMatchers(HttpMethod.PUT, "/api/espacios/**").hasRole("ADMIN")
                        .requestMatchers(HttpMethod.DELETE, "/api/espacios/**").hasRole("ADMIN")
```

```java
import java.time.Clock;
import java.time.Instant;

@Service
public class JwtService {

    private final JwtEncoder encoder;
    private final JwtProperties propiedades;
    private final Clock clock;

    public JwtService(JwtEncoder encoder, JwtProperties propiedades, Clock clock) {
        this.encoder = encoder;
        this.propiedades = propiedades;
        this.clock = clock;
    }

    public TokenResponse generar(Usuario usuario) {
        Instant ahora = clock.instant();
        Instant expira = ahora.plus(propiedades.expiracion());
        JwtClaimsSet claims = JwtClaimsSet.builder()
                .issuer("coworking-booking")
                .issuedAt(ahora)
                .expiresAt(expira)
                .subject(usuario.getId().toString())
                .claim("email", usuario.getEmail())
                .claim("rol", usuario.getRol().name())
                .build();
        JwsHeader header = JwsHeader.with(MacAlgorithm.HS256).build();
        String token = encoder.encode(JwtEncoderParameters.from(header, claims)).getTokenValue();
        return new TokenResponse(token, expira);
    }
}
```

```java
package co.devdavco.coworking.auth;

import co.devdavco.coworking.auth.dto.LoginRequest;
import co.devdavco.coworking.auth.dto.RegistroRequest;
import co.devdavco.coworking.auth.dto.TokenResponse;
import co.devdavco.coworking.common.ConflictoException;
import co.devdavco.coworking.common.CredencialesInvalidasException;
import co.devdavco.coworking.usuario.Rol;
import co.devdavco.coworking.usuario.Usuario;
import co.devdavco.coworking.usuario.UsuarioRepository;
import co.devdavco.coworking.usuario.dto.UsuarioResponse;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.nio.charset.StandardCharsets;
import java.util.Locale;

@Service
public class AuthService {

    private static final int LIMITE_BYTES_BCRYPT = 72;

    private final UsuarioRepository usuarios;
    private final PasswordEncoder passwordEncoder;
    private final JwtService jwtService;

    public AuthService(UsuarioRepository usuarios, PasswordEncoder passwordEncoder, JwtService jwtService) {
        this.usuarios = usuarios;
        this.passwordEncoder = passwordEncoder;
        this.jwtService = jwtService;
    }

    @Transactional
    public UsuarioResponse registrar(RegistroRequest request) {
        String email = normalizarEmail(request.email());
        if (usuarios.existsByEmail(email)) {
            throw new ConflictoException("Ya existe una cuenta con el email " + email);
        }
        Usuario usuario = new Usuario(request.nombre().trim(), email,
                passwordEncoder.encode(request.password()), Rol.USUARIO);
        return UsuarioResponse.desde(usuarios.save(usuario));
    }

    @Transactional(readOnly = true)
    public TokenResponse login(LoginRequest request) {
        if (request.password().getBytes(StandardCharsets.UTF_8).length > LIMITE_BYTES_BCRYPT) {
            throw new CredencialesInvalidasException();
        }
        Usuario usuario = usuarios.findByEmail(normalizarEmail(request.email()))
                .filter(encontrado -> passwordEncoder.matches(request.password(), encontrado.getPasswordHash()))
                .orElseThrow(CredencialesInvalidasException::new);
        return jwtService.generar(usuario);
    }

    public static String normalizarEmail(String email) {
        return email.trim().toLowerCase(Locale.ROOT);
    }
}
```

```java
package co.devdavco.coworking.usuario;

import co.devdavco.coworking.common.RecursoNoEncontradoException;
import co.devdavco.coworking.usuario.dto.UsuarioResponse;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.List;

@Service
@Transactional(readOnly = true)
public class UsuarioService {

    private final UsuarioRepository usuarios;

    public UsuarioService(UsuarioRepository usuarios) {
        this.usuarios = usuarios;
    }

    public UsuarioResponse obtener(Long id) {
        return usuarios.findById(id)
                .map(UsuarioResponse::desde)
                .orElseThrow(() -> new RecursoNoEncontradoException("No existe el usuario " + id));
    }

    public List<UsuarioResponse> listar() {
        return usuarios.findAllByOrderByNombreAsc().stream()
                .map(UsuarioResponse::desde)
                .toList();
    }
}
```

```java
package co.devdavco.coworking.auth;

import co.devdavco.coworking.usuario.Rol;
import co.devdavco.coworking.usuario.Usuario;
import co.devdavco.coworking.usuario.UsuarioRepository;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.boot.ApplicationArguments;
import org.springframework.boot.ApplicationRunner;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.stereotype.Component;

/** Al arrancar, crea el ADMIN definido en ADMIN_EMAIL / ADMIN_PASSWORD si todavía no existe. */
@Component
public class AdminInicial implements ApplicationRunner {

    private static final Logger log = LoggerFactory.getLogger(AdminInicial.class);

    private final AdminProperties propiedades;
    private final UsuarioRepository usuarios;
    private final PasswordEncoder passwordEncoder;

    public AdminInicial(AdminProperties propiedades, UsuarioRepository usuarios, PasswordEncoder passwordEncoder) {
        this.propiedades = propiedades;
        this.usuarios = usuarios;
        this.passwordEncoder = passwordEncoder;
    }

    @Override
    public void run(ApplicationArguments args) {
        if (vacio(propiedades.email()) || vacio(propiedades.password())) {
            log.info("ADMIN_EMAIL o ADMIN_PASSWORD no definidos: no se crea administrador inicial");
            return;
        }
        String email = AuthService.normalizarEmail(propiedades.email());
        if (usuarios.existsByEmail(email)) {
            return;
        }
        usuarios.save(new Usuario("Administrador", email, passwordEncoder.encode(propiedades.password()), Rol.ADMIN));
        log.info("Administrador inicial creado: {}", email);
    }

    private static boolean vacio(String valor) {
        return valor == null || valor.isBlank();
    }
}
```

```java
package co.devdavco.coworking.reserva;

import co.devdavco.coworking.common.ReglaNegocioException;
import org.springframework.stereotype.Component;

import java.time.Clock;
import java.time.Duration;
import java.time.Instant;
import java.time.ZonedDateTime;

/** Reglas de horario de una reserva. No toca la base de datos: se prueba con un reloj fijo. */
@Component
public class ReglasHorario {

    private final ReservasProperties propiedades;
    private final Clock clock;

    public ReglasHorario(ReservasProperties propiedades, Clock clock) {
        this.propiedades = propiedades;
        this.clock = clock;
    }

    public void validar(Instant inicio, Instant fin) {
        if (!inicio.isAfter(clock.instant())) {
            throw new ReglaNegocioException("La reserva debe empezar en el futuro");
        }
        if (!fin.isAfter(inicio)) {
            throw new ReglaNegocioException("La hora de fin debe ser posterior a la de inicio");
        }
        Duration duracion = Duration.between(inicio, fin);
        if (duracion.compareTo(propiedades.duracionMinima()) < 0) {
            throw new ReglaNegocioException(
                    "La reserva debe durar al menos " + propiedades.duracionMinima().toMinutes() + " minutos");
        }
        if (duracion.compareTo(propiedades.duracionMaxima()) > 0) {
            throw new ReglaNegocioException(
                    "La reserva no puede durar más de " + propiedades.duracionMaxima().toMinutes() + " minutos");
        }
        ZonedDateTime inicioLocal = inicio.atZone(propiedades.zonaHoraria());
        ZonedDateTime finLocal = fin.atZone(propiedades.zonaHoraria());
        if (!alineado(inicioLocal) || !alineado(finLocal)) {
            throw new ReglaNegocioException(
                    "Las horas deben ir en bloques de " + propiedades.bloque().toMinutes() + " minutos");
        }
        if (!inicioLocal.toLocalDate().equals(finLocal.toLocalDate())) {
            throw new ReglaNegocioException("La reserva debe empezar y terminar el mismo día");
        }
        if (inicioLocal.toLocalTime().isBefore(propiedades.apertura())
                || finLocal.toLocalTime().isAfter(propiedades.cierre())) {
            throw new ReglaNegocioException(
                    "La reserva debe estar entre " + propiedades.apertura() + " y " + propiedades.cierre());
        }
    }

    public Instant calcularFinTotal(Instant fin, int minutosLimpieza) {
        return fin.plus(Duration.ofMinutes(minutosLimpieza));
    }

    private boolean alineado(ZonedDateTime momento) {
        return momento.getNano() == 0
                && momento.toLocalTime().toSecondOfDay() % propiedades.bloque().toSeconds() == 0;
    }
}
```

```java
package co.devdavco.coworking.reserva;

import co.devdavco.coworking.common.ConflictoException;
import co.devdavco.coworking.espacio.Espacio;
import co.devdavco.coworking.espacio.EspacioService;
import co.devdavco.coworking.reserva.dto.CrearReservaRequest;
import co.devdavco.coworking.reserva.dto.ReservaResponse;
import co.devdavco.coworking.usuario.Usuario;
import co.devdavco.coworking.usuario.UsuarioRepository;
import org.springframework.dao.DataIntegrityViolationException;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.time.Clock;
import java.time.Instant;
import java.time.format.DateTimeFormatter;
import java.util.List;

@Service
public class ReservaService {

    private static final String CONSTRAINT_SOLAPAMIENTO = "reservas_sin_solapamiento";
    private static final DateTimeFormatter HORA = DateTimeFormatter.ofPattern("HH:mm");

    private final ReservaRepository reservas;
    private final EspacioService espacioService;
    private final UsuarioRepository usuarios;
    private final ReglasHorario reglasHorario;
    private final ReservasProperties propiedades;
    private final Clock clock;

    public ReservaService(ReservaRepository reservas, EspacioService espacioService, UsuarioRepository usuarios,
                          ReglasHorario reglasHorario, ReservasProperties propiedades, Clock clock) {
        this.reservas = reservas;
        this.espacioService = espacioService;
        this.usuarios = usuarios;
        this.reglasHorario = reglasHorario;
        this.propiedades = propiedades;
        this.clock = clock;
    }

    @Transactional
    public ReservaResponse crear(Long usuarioId, CrearReservaRequest request) {
        Espacio espacio = espacioService.buscarActivo(request.espacioId());
        Instant inicio = request.horaInicio().toInstant();
        Instant fin = request.horaFin().toInstant();
        reglasHorario.validar(inicio, fin);
        Instant finTotal = reglasHorario.calcularFinTotal(fin, espacio.getMinutosLimpieza());

        List<Reserva> cruces = reservas.findSolapadas(espacio.getId(), EstadoReserva.CONFIRMADA, inicio, finTotal);
        if (!cruces.isEmpty()) {
            throw franjaOcupada(cruces.get(0));
        }

        Usuario usuario = usuarios.getReferenceById(usuarioId);
        Reserva reserva = new Reserva(usuario, espacio, inicio, fin, finTotal, clock.instant());
        try {
            reserva = reservas.saveAndFlush(reserva);
        } catch (DataIntegrityViolationException ex) {
            // Dos peticiones pasaron la consulta a la vez: la base de datos rechazó la segunda.
            if (esSolapamiento(ex)) {
                throw new ConflictoException("El espacio ya está ocupado en ese horario");
            }
            throw ex;
        }
        return mapear(reserva);
    }

    private ReservaResponse mapear(Reserva reserva) {
        return ReservaResponse.desde(reserva, propiedades.zonaHoraria(), clock.instant());
    }

    private ConflictoException franjaOcupada(Reserva existente) {
        String desde = HORA.format(existente.getHoraInicio().atZone(propiedades.zonaHoraria()));
        String hasta = HORA.format(existente.getHoraFinTotal().atZone(propiedades.zonaHoraria()));
        return new ConflictoException(
                "El espacio ya está ocupado de " + desde + " a " + hasta + " (incluye limpieza)");
    }

    private static boolean esSolapamiento(DataIntegrityViolationException ex) {
        String mensaje = ex.getMostSpecificCause().getMessage();
        return mensaje != null && mensaje.contains(CONSTRAINT_SOLAPAMIENTO);
    }
}
```

```java
    public List<ReservaResponse> listarMias(Long usuarioId) {
        return reservas.findAllByUsuario_IdOrderByHoraInicioDesc(usuarioId).stream()
                .map(this::mapear)
                .toList();
    }

    @Transactional(readOnly = true)
    public List<ReservaResponse> listar(Long espacioId, LocalDate fecha) {
        List<Specification<Reserva>> filtros = new ArrayList<>();
        if (espacioId != null) {
            filtros.add(ReservaSpecs.delEspacio(espacioId));
        }
        if (fecha != null) {
            filtros.add(ReservaSpecs.seCruzaCon(inicioDelDia(fecha), inicioDelDia(fecha.plusDays(1))));
        }
        return reservas.findAll(Specification.allOf(filtros), Sort.by(Sort.Direction.DESC, "horaInicio")).stream()
                .map(this::mapear)
                .toList();
    }

    @Transactional(readOnly = true)
    public ReservaResponse obtener(UsuarioActual actual, Long id) {
        return mapear(buscarVisible(actual, id));
    }

    @Transactional
    public ReservaResponse cancelar(UsuarioActual actual, Long id) {
        Reserva reserva = buscarVisible(actual, id);
        if (reserva.getEstado() == EstadoReserva.CANCELADA) {
            throw new ConflictoException("La reserva ya está cancelada");
        }
        if (!actual.esAdmin() && !reserva.getHoraInicio().isAfter(clock.instant())) {
            throw new ReglaNegocioException("No puedes cancelar una reserva que ya empezó");
        }
        reserva.cancelar();
        return mapear(reserva);
    }

    @Transactional(readOnly = true)
    public List<FranjaOcupada> ocupacion(Long espacioId, LocalDate fecha) {
        espacioService.buscarActivo(espacioId);
        return reservas.findSolapadas(espacioId, EstadoReserva.CONFIRMADA,
                        inicioDelDia(fecha), inicioDelDia(fecha.plusDays(1))).stream()
                .map(reserva -> new FranjaOcupada(
                        reserva.getHoraInicio().atZone(propiedades.zonaHoraria()).toOffsetDateTime(),
                        reserva.getHoraFinTotal().atZone(propiedades.zonaHoraria()).toOffsetDateTime()))
                .toList();
    }

    /** Un usuario solo ve sus reservas; si pide la de otro, responde 404 para no revelar que existe. */
    private Reserva buscarVisible(UsuarioActual actual, Long id) {
        return reservas.findById(id)
                .filter(reserva -> actual.esAdmin() || reserva.esDe(actual.id()))
                .orElseThrow(() -> new RecursoNoEncontradoException("No existe la reserva " + id));
    }

    private Instant inicioDelDia(LocalDate fecha) {
        return fecha.atStartOfDay(propiedades.zonaHoraria()).toInstant();
    }
```
